cmake_minimum_required(VERSION 3.16)
project(PlenarioFighter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(PlenarioFighter src/main.cpp)
if(WIN32 AND MINGW)
    target_include_directories(PlenarioFighter PRIVATE vendor/include/SDL2)
    target_link_directories(PlenarioFighter PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/lib")
    target_link_libraries(PlenarioFighter PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/lib/libSDL2.a" dinput8 dxguid dxerr8 user32 gdi32 winmm imm32 ole32 oleaut32 shell32 setupapi version uuid)
    target_link_options(PlenarioFighter PRIVATE -static -static-libgcc -static-libstdc++)
else()
    find_package(SDL2 CONFIG REQUIRED)
    target_link_libraries(PlenarioFighter PRIVATE SDL2::SDL2)
endif()
add_custom_command(TARGET PlenarioFighter POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/gamecontrollerdb.txt" "$<TARGET_FILE_DIR:PlenarioFighter>")
enable_testing()
add_test(NAME Combat COMMAND PlenarioFighter --self-test)
add_test(NAME CombosAndAudio COMMAND PlenarioFighter --combo-test)
add_test(NAME DifficultyAmmoKO COMMAND PlenarioFighter --combat-v12-test)
add_test(NAME MenusAndEndings COMMAND PlenarioFighter --flow-test)
add_test(NAME ArcadeAndPain COMMAND PlenarioFighter --arcade-test)
add_custom_command(TARGET PlenarioFighter POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/audio" "$<TARGET_FILE_DIR:PlenarioFighter>/audio")
add_test(NAME Controllers COMMAND PlenarioFighter --input-test)

add_test(NAME BonusAndMusic COMMAND PlenarioFighter --bonus-test)
add_test(NAME PresentationAndReporter COMMAND PlenarioFighter --presentation-test)

add_test(NAME ExpandedRosterAndHistory COMMAND PlenarioFighter --expansion-test)
